# Exclusion guards for title keywords (Odesa region). One regex per line, matched
# against the lower-cased title with apostrophes normalized to '.
# A guarded phrase is blanked out before keyword matching, so a title still
# matches on any other keyword (e.g. "Одеса" elsewhere in the same title).
# Keep every guard narrow: it must never hide real Odesa-region news.

# Kyiv's Подільський район, Kamianets-Podilskyi and Podillia (Khmelnytskyi / Vinnytsia)
подільськ\w*\s+район\w*(?:\s+(?:міста|м\.))?\s+(?:києва|столиці)
(?:києв\w*|столиц\w*)[^.!?]{0,40}?подільськ\w*\s+район\w*
кам'янц?\w*-подільськ\w*
подільськ\w*[^.!?]{0,40}?(?:хмельниц\w*|хмельниччин\w*|вінниц\w*|вінниччин\w*)
(?:хмельниц\w*|хмельниччин\w*|вінниц\w*|вінниччин\w*)[^.!?]{0,40}?подільськ\w*

# "Південн*" as a generic adjective, not the town Південне / port Південний
південн\w*\s+(?:коре\w*|буг\w*|бузі|кавказ\w*|африк\w*|судан\w*|америк\w*|полюс\w*|осет\w*|ліван\w*)
південн\w*\s+(?:фронт\w*|напрям\w*|оперативн\w*|угрупован\w*|сил\w*\s+оборони)
південн\w*\s+(?:регіон\w*|област\w*|частин\w*|україн\w*|узбережж\w*|кордон\w*|півкул\w*|сусід\w*|схід\w*|сход\w*|захід\w*|заход\w*)
південн\w*\s+(?:міст\w*|вокзал\w*|залізниц\w*)\s*(?:у\s+|в\s+)?(?:києв\w*|харков\w*|харкові)?
південн\w*\s+(?:залізниц\w*)
# Південне in Kharkiv oblast
південн\w*[^.!?]{0,40}?харківщин\w*
харківщин\w*[^.!?]{0,40}?південн\w*

# Чорноморський флот / Crimea
чорноморськ\w*\s+флот\w*
чорноморськ\w*[^.!?]{0,30}?(?:крим\w*|кримськ\w*)

# "роздільна здатність" (resolution), not the town Роздільна
роздільн\w*\s+здатн\w*

# Same town names in other oblasts
(?:березів[кц]\w*|біляїв[кц]\w*|балт\w*)[^.!?]{0,40}?(?:житомирщин\w*|київщин\w*|харківщин\w*|миколаївщин\w*|черкащин\w*|чернігівщин\w*|полтавщин\w*|дніпропетровщин\w*|сумщин\w*|кіровоградщин\w*|херсонщин\w*|запоріжж\w*|львівщин\w*|вінниччин\w*|хмельниччин\w*)
