# Exclusion guards for title keywords (Kharkiv region). One regex per line,
# matched against the lower-cased title with apostrophes normalized to '.
# A guarded phrase is blanked out before keyword matching, so a title still
# matches on any other, unguarded keyword (e.g. "Харків" in the same title).
# Keep every guard narrow: it must never hide real Kharkiv-oblast news.

# "Південн*" as a generic adjective, not the town Південне / Південна громада
південн\w*\s+(?:коре\w*|буг\w*|бузі|кавказ\w*|африк\w*|судан\w*|америк\w*|полюс\w*|осет\w*|ліван\w*)
південн\w*\s+(?:фронт\w*|напрям\w*|оперативн\w*|угрупован\w*|сил\w*\s+оборони)
південн\w*\s+(?:регіон\w*|област\w*|частин\w*|україн\w*|узбережж\w*|кордон\w*|півкул\w*|сусід\w*|схід\w*|сход\w*|захід\w*|заход\w*)
південн\w*\s+(?:вокзал\w*|залізниц\w*)
південн\w*\s+залізниц\w*

# Київський район of the city of Kyiv, not Kharkiv's Київський район
київськ\w*\s+район\w*(?:\s+(?:міста|м\.))?\s+(?:києва|столиці)
(?:києв\w*|столиц\w*)[^.!?]{0,40}?київськ\w*\s+район\w*

# Шевченківський район in other cities (Kyiv, Zaporizhzhia, Lviv, …)
шевченківськ\w*\s+район\w*[^.!?]{0,40}?(?:києва|столиці|львов\w*|львів\w*|запоріж\w*|дніпр\w*|полтав\w*|черкас\w*|миколаїв\w*|одес\w*|вінниц\w*)
(?:києва|столиці|львов\w*|львів\w*|запоріж\w*|дніпр\w*|полтав\w*|черкас\w*|миколаїв\w*|одес\w*|вінниц\w*)[^.!?]{0,40}?шевченківськ\w*\s+район\w*

# Індустріальний as a generic adjective (парк, зона, район other cities)
індустріальн\w*\s+(?:парк\w*|зон\w*|підприємств\w*|гігант\w*|район\w*\s+(?:міста|м\.)?\s*(?:києва|дніпр\w*|запоріж\w*))

# Золочів / Золочівська громада in Lviv or Kyiv oblast
(?:золочівськ\w*|золочів\w*)[^.!?]{0,40}?(?:львівщин\w*|львівськ\w*|львов\w*|київщин\w*|київськ\w*\s+област\w*|замок\w*)
(?:львівщин\w*|львівськ\w*|львов\w*|київщин\w*|київськ\w*\s+област\w*)[^.!?]{0,40}?(?:золочівськ\w*|золочів\w*)

# Біляївка / Біляївська громада in Odesa oblast
(?:біляївськ\w*|біляївк\w*)[^.!?]{0,40}?(?:одещин\w*|одеськ\w*|одес\w*)
(?:одещин\w*|одеськ\w*|одес\w*)[^.!?]{0,40}?(?:біляївськ\w*|біляївк\w*)

# Слобожанська громада / Слобожанське in Dnipropetrovsk oblast
(?:слобожанськ\w*)[^.!?]{0,40}?(?:дніпропетровщин\w*|дніпровськ\w*|дніпрі|дніпра|самарівськ\w*)
(?:дніпропетровщин\w*|дніпровськ\w*|самарівськ\w*)[^.!?]{0,40}?слобожанськ\w*

# Золочівська already covered. Other same-name hromadas used with another oblast
(?:коломацьк\w*|коломак\w*)[^.!?]{0,40}?(?:полтавщин\w*|полтавськ\w*)
(?:полтавщин\w*|полтавськ\w*)[^.!?]{0,40}?(?:коломацьк\w*|коломак\w*)

(?:височанськ\w*)[^.!?]{0,40}?(?:чернігівщин\w*|чернігівськ\w*)
(?:чернігівщин\w*|чернігівськ\w*)[^.!?]{0,40}?височанськ\w*

(?:малинівськ\w*)[^.!?]{0,40}?(?:запорізьк\w*|запоріжж\w*)
(?:запорізьк\w*|запоріжж\w*)[^.!?]{0,40}?малинівськ\w*

(?:новопокровськ\w*)[^.!?]{0,40}?(?:дніпропетровщин\w*|дніпровськ\w*|крим\w*)
(?:дніпропетровщин\w*|дніпровськ\w*|крим\w*)[^.!?]{0,40}?новопокровськ\w*

(?:олексіївськ\w*)[^.!?]{0,40}?(?:крим\w*)
(?:крим\w*)[^.!?]{0,40}?олексіївськ\w*

(?:петропавлівськ\w*)[^.!?]{0,40}?(?:дніпропетровщин\w*|дніпровськ\w*|миколаївщин\w*|миколаївськ\w*)
(?:дніпропетровщин\w*|дніпровськ\w*|миколаївщин\w*|миколаївськ\w*)[^.!?]{0,40}?петропавлівськ\w*

# Лозова / Валки / Барвінкове as villages in other oblasts (rare in our sources)
(?:лозов\w*|валк\w*|барвінков\w*|златопіль\w*|люботин\w*)[^.!?]{0,40}?(?:вінниччин\w*|хмельниччин\w*|сумщин\w*|чернігівщин\w*|дніпропетровщин\w*|полтавщин\w*|житомирщин\w*|кіровоградщин\w*|черкащин\w*)
(?:вінниччин\w*|хмельниччин\w*|сумщин\w*|чернігівщин\w*|дніпропетровщин\w*|полтавщин\w*|житомирщин\w*|кіровоградщин\w*|черкащин\w*)[^.!?]{0,40}?(?:лозов\w*|валк\w*|барвінков\w*|златопіль\w*|люботин\w*)
